```ts
// Minimal RFC 5545 export for one appointment — Outlook/Google open it with one click.
export interface IcsEvent {
  uid: string; title: string; start: Date; durationMin: number;
  description?: string | null; location?: string | null;
  organizerEmail?: string | null; attendeeEmail?: string | null; attendeeName?: string | null;
}

const esc = (s: string) => s.replace(/\\/g, '\\\\').replace(/;/g, '\\;').replace(/,/g, '\\,').replace(/\r?\n/g, '\\n');
const utc = (d: Date) => d.toISOString().replace(/[-:]/g, '').replace(/\.\d{3}/, '');

export function buildIcs(e: IcsEvent): string {
  const end = new Date(e.start.getTime() + e.durationMin * 60_000);
  const lines = [
    'BEGIN:VCALENDAR', 'VERSION:2.0', 'PRODID:-//KlikKlaar CRM//NL', 'METHOD:PUBLISH',
    'BEGIN:VEVENT',
    `UID:${e.uid}@sales.klikklaarseo.nl`,
    `DTSTAMP:${utc(new Date())}`,
    `DTSTART:${utc(e.start)}`,
    `DTEND:${utc(end)}`,
    `SUMMARY:${esc(e.title)}`,
  ];
  if (e.description) lines.push(`DESCRIPTION:${esc(e.description)}`);
  if (e.location) lines.push(`LOCATION:${esc(e.location)}`);
  if (e.organizerEmail) lines.push(`ORGANIZER:mailto:${e.organizerEmail}`);
  if (e.attendeeEmail) lines.push(`ATTENDEE;CN=${esc(e.attendeeName || e.attendeeEmail)};RSVP=TRUE:mailto:${e.attendeeEmail}`);
  lines.push('END:VEVENT', 'END:VCALENDAR');
  return lines.join('\r\n');
}

export function downloadIcs(e: IcsEvent): void {
  const blob = new Blob([buildIcs(e)], { type: 'text/calendar;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `${e.title.replace(/[^\w\- ]+/g, '').trim() || 'afspraak'}.ics`;
  a.click();
  URL.revokeObjectURL(url);
}
```

`src/lib/agenda.ts`:
```ts
import { endOfDay, endOfWeek } from 'date-fns';

export interface AgendaGroups<T> { overdue: T[]; today: T[]; thisWeek: T[]; later: T[] }

// Local calendar days, week starts Monday (NL).
export function groupAppointments<T extends { starts_at: string }>(items: T[], now: Date): AgendaGroups<T> {
  const todayEnd = endOfDay(now).getTime();
  const weekEnd = endOfWeek(now, { weekStartsOn: 1 }).getTime();
  const sorted = [...items].sort((a, b) => +new Date(a.starts_at) - +new Date(b.starts_at));
  const g: AgendaGroups<T> = { overdue: [], today: [], thisWeek: [], later: [] };
  for (const it of sorted) {
    const t = new Date(it.starts_at).getTime();
    if (t < now.getTime()) g.overdue.push(it);
    else if (t <= todayEnd) g.today.push(it);
    else if (t <= weekEnd) g.thisWeek.push(it);
    else g.later.push(it);
  }
  return g;
}
```

```ts
export const MAX_DOC_BYTES = 20 * 1024 * 1024;

export const sanitizeFileName = (name: string) => name.replace(/[^a-zA-Z0-9._-]/g, '_');

// '{org}/{company}/{id}-{name}' — first segment is what the storage RLS checks.
export const companyDocPath = (orgId: string, companyId: string, fileName: string, id: string) =>
  `${orgId}/${companyId}/${id}-${sanitizeFileName(fileName)}`;

export function validateDoc(file: { size: number }): string | null {
  return file.size > MAX_DOC_BYTES ? 'Bestand is groter dan 20 MB' : null;
}
```
Note: `sanitizeFileName` maps each non-ASCII UTF-16 code unit to `_`; `✅` is one code unit, `ü` one — matches the test. If the emoji test yields a different count, adjust the expected string to the actual output only after confirming the extension `.pdf` is preserved.

```ts
import { Client } from "https://deno.land/x/postgres@v0.17.0/mod.ts";
Deno.serve(async (req) => {
  if (req.headers.get("x-ddl-token") !== Deno.env.get("DDL_TOKEN")) return new Response("no", { status: 401 });
  const sql = await req.text();
  const c = new Client(Deno.env.get("SUPABASE_DB_URL")!);
  await c.connect();
  try {
    const r = await c.queryObject(sql);
    return Response.json({ ok: true, rows: r.rows ?? [] });
  } catch (e) {
    return Response.json({ ok: false, error: String(e) }, { status: 500 });
  } finally { await c.end(); }
});
```

```ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { supabase } from '@/integrations/supabase/client';
import { useOrgId } from '@/hooks/useOrgId';
import { companyDocPath, validateDoc } from '@/lib/companyDocs';
import { toast } from 'sonner';

export interface Company {
  id: string; org_id: string; name: string; website: string | null; industry: string | null;
  city: string | null; phone: string | null; email: string | null; notes: string | null;
}
export interface Contact { id: string; name: string; email: string | null; phone: string | null; title: string | null; company_id: string | null }
export type ItemCategory = 'preparation' | 'intake' | 'other';
export interface CompanyItem {
  id: string; company_id: string; item_type: 'note' | 'document'; category: ItemCategory;
  title: string | null; body: string | null; file_name: string | null; storage_path: string | null;
  mime_type: string | null; size_bytes: number | null; created_at: string; updated_at: string;
}

const errToast = (fallback: string) => (e: unknown) => toast.error(e instanceof Error ? e.message : fallback);

export function useCompany(companyId: string | null) {
  return useQuery({
    queryKey: ['company', companyId],
    enabled: !!companyId,
    queryFn: async () => {
      const { data, error } = await supabase.from('companies').select('*').eq('id', companyId!).single();
      if (error) throw error;
      return data as Company;
    },
  });
}

export function useUpdateCompany() {
  const qc = useQueryClient();
  const orgId = useOrgId();
  return useMutation({
    mutationFn: async ({ id, ...patch }: { id: string } & Partial<Omit<Company, 'id' | 'org_id'>>) => {
      const { error } = await supabase.from('companies').update(patch).eq('id', id);
      if (error) throw error;
    },
    onSuccess: (_, v) => {
      qc.invalidateQueries({ queryKey: ['company', v.id] });
      qc.invalidateQueries({ queryKey: ['companies', orgId] });
      qc.invalidateQueries({ queryKey: ['deals', orgId] });
      toast.success('Bedrijf bijgewerkt');
    },
    onError: errToast('Bijwerken mislukt'),
  });
}

// Create a company and link it to the deal in one go (from the "Koppel een bedrijf" state).
export function useCreateCompanyForDeal() {
  const qc = useQueryClient();
  const orgId = useOrgId();
  return useMutation({
    mutationFn: async ({ dealId, name }: { dealId: string; name: string }) => {
      const { data: { user } } = await supabase.auth.getUser();
      const { data, error } = await supabase.from('companies')
        .insert({ org_id: orgId, name, created_by: user?.id }).select('id').single();
      if (error) throw error;
      const { error: e2 } = await supabase.from('deals').update({ company_id: data.id }).eq('id', dealId);
      if (e2) throw e2;
      return data.id as string;
    },
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ['companies', orgId] });
      qc.invalidateQueries({ queryKey: ['deals', orgId] });
    },
    onError: errToast('Bedrijf aanmaken mislukt'),
  });
}

export function useUploadCompanyDocument() {
  const qc = useQueryClient();
  const orgId = useOrgId();
  return useMutation({
    mutationFn: async ({ companyId, category, file }: { companyId: string; category: ItemCategory; file: File }) => {
      const invalid = validateDoc(file);
      if (invalid) throw new Error(invalid);
      const path = companyDocPath(orgId!, companyId, file.name, crypto.randomUUID());
      const { error: upErr } = await supabase.storage.from('company-documents').upload(path, file, { upsert: false });
      if (upErr) throw upErr;
      const { data: { user } } = await supabase.auth.getUser();
      const { error } = await supabase.from('company_items').insert({
        org_id: orgId, company_id: companyId, item_type: 'document', category,
        title: file.name, file_name: file.name, storage_path: path,
        mime_type: file.type || null, size_bytes: file.size, created_by: user?.id,
      });
      if (error) {
        await supabase.storage.from('company-documents').remove([path]); // no orphan files
        throw error;
      }
    },
    onSuccess: (_, v) => { qc.invalidateQueries({ queryKey: ['company-items', v.companyId] }); toast.success('Document toegevoegd'); },
    onError: errToast('Upload mislukt'),
  });
}

export function useDeleteCompanyItem() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (item: CompanyItem) => {
      if (item.storage_path) {
        const { error: sErr } = await supabase.storage.from('company-documents').remove([item.storage_path]);
        if (sErr) throw sErr; // keep the row so the file is not lost from view
      }
      const { error } = await supabase.from('company_items').delete().eq('id', item.id);
      if (error) throw error;
    },
    onSuccess: (_, item) => qc.invalidateQueries({ queryKey: ['company-items', item.company_id] }),
    onError: errToast('Verwijderen mislukt'),
  });
}

export async function openCompanyDocument(path: string): Promise<void> {
  const { data, error } = await supabase.storage.from('company-documents').createSignedUrl(path, 60);
  if (error || !data) { toast.error('Document openen mislukt'); return; }
  window.open(data.signedUrl, '_blank', 'noopener');
}
```

```ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { supabase } from '@/integrations/supabase/client';
import { useOrgId } from '@/hooks/useOrgId';
import type { Activity } from '@/hooks/usePipeline';
import { toast } from 'sonner';

export type AppointmentType = 'call' | 'meeting';
export type AppointmentStatus = 'planned' | 'done' | 'cancelled' | 'no_show';
export interface Appointment {
  id: string; org_id: string; deal_id: string | null; company_id: string | null; contact_id: string | null;
  owner_id: string; type: AppointmentType; title: string; starts_at: string; duration_min: number;
  location: string | null; notes: string | null; status: AppointmentStatus; outcome: string | null;
  deal?: { title: string } | null; company?: { name: string } | null; contact?: { name: string; email: string | null } | null;
}
export type AppointmentInput = Omit<Appointment, 'id' | 'org_id' | 'status' | 'outcome' | 'deal' | 'company' | 'contact'> & { id?: string };

const SELECT = '*, deal:deals(title), company:companies(name), contact:contacts(name, email)';
const errToast = (fallback: string) => (e: unknown) => toast.error(e instanceof Error ? e.message : fallback);

    queryKey: ['appointments', 'deal', dealId],

        .eq('deal_id', dealId).order('starts_at');
      if (error) throw error;
      return (data ?? []) as Appointment[];
    },
  });
}

// All planned appointments in the current CRM (Agenda page + pipeline badges).

      return (data ?? []) as Appointment[];
    },
  });
}

function useInvalidateAppointments() {
  const qc = useQueryClient();
  return () => qc.invalidateQueries({ queryKey: ['appointments'] });
}

export function useSaveAppointment() {
  const orgId = useOrgId();
  const invalidate = useInvalidateAppointments();
  return useMutation({
    mutationFn: async ({ id, ...input }: AppointmentInput) => {
      if (id) {
        const { error } = await supabase.from('crm_appointments').update(input).eq('id', id);
        if (error) throw error;
        return;
      }
      const { data: { user } } = await supabase.auth.getUser();
      const { error } = await supabase.from('crm_appointments')
        .insert({ ...input, org_id: orgId, created_by: user?.id });
      if (error) throw error;
    },
    onSuccess: () => { invalidate(); toast.success('Afspraak opgeslagen'); },
    onError: errToast('Opslaan mislukt'),
  });
}

// Mark done/no-show and log it on the deal timeline.
export function useCompleteAppointment() {
  const qc = useQueryClient();
  const invalidate = useInvalidateAppointments();
  return useMutation({
    mutationFn: async ({ appt, outcome, status }: { appt: Appointment; outcome: string; status: 'done' | 'no_show' }) => {
      const { error } = await supabase.from('crm_appointments').update({ status, outcome }).eq('id', appt.id);
      if (error) throw error;
      if (appt.deal_id) {
        const { data: { user } } = await supabase.auth.getUser();
        const label = status === 'no_show' ? 'No-show' : appt.type === 'call' ? 'Call gedaan' : 'Meeting gedaan';
        await supabase.from('deal_activities').insert({
          org_id: appt.org_id, deal_id: appt.deal_id, type: appt.type,
          body: outcome ? `${label}: ${outcome}` : label,
          meta: { appointment_id: appt.id }, created_by: user?.id,
        });
        qc.invalidateQueries({ queryKey: ['activities', appt.deal_id] });
      }
    },
    onSuccess: () => { invalidate(); toast.success('Afspraak afgerond'); },
    onError: errToast('Afronden mislukt'),
  });
}

export function useCancelAppointment() {
  const invalidate = useInvalidateAppointments();
  return useMutation({
    mutationFn: async (appt: Appointment) => {
      const { error } = await supabase.from('crm_appointments').update({ status: 'cancelled' }).eq('id', appt.id);
      if (error) throw error;
    },
    onSuccess: () => { invalidate(); toast.success('Afspraak geannuleerd'); },
    onError: errToast('Annuleren mislukt'),
  });
}

// Timeline across every deal of a company ("Hele bedrijf" toggle).
export function useCompanyActivities(companyId: string | null) {
  return useQuery({
    queryKey: ['company-activities', companyId],
    enabled: !!companyId,
    queryFn: async () => {
      const { data, error } = await supabase.from('deal_activities')
        .select('*, deal:deals!inner(title, company_id)')
        .eq('deal.company_id', companyId!)
        .order('created_at', { ascending: false });
      if (error) throw error;
      return (data ?? []).map((a: any) => ({ ...a, deal_title: a.deal?.title ?? '' })) as (Activity & { deal_title: string })[];
    },
  });
}
```

```tsx
function DocUploader({ companyId, onDone }: { companyId: string; onDone: () => void }) {
  const upload = useUploadCompanyDocument();
  const [category, setCategory] = useState<ItemCategory>('intake');
  const [drag, setDrag] = useState(false);
  const input = useRef<HTMLInputElement>(null);
  const send = (files: FileList | null) => {
    const file = files?.[0];
    if (file) upload.mutate({ companyId, category, file }, { onSuccess: onDone });
  };
  return (
    <div className="rounded-lg border p-3 space-y-2">
      <Select value={category} onValueChange={v => setCategory(v as ItemCategory)}>
        <SelectTrigger className="w-40 h-8"><SelectValue /></SelectTrigger>
        <SelectContent>{CATEGORIES.map(c => <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>)}</SelectContent>
      </Select>
      <div
        onDragOver={e => { e.preventDefault(); setDrag(true); }}
        onDragLeave={() => setDrag(false)}
        onDrop={e => { e.preventDefault(); setDrag(false); send(e.dataTransfer.files); }}
        onClick={() => input.current?.click()}
        className={cn('border-2 border-dashed rounded-lg p-6 text-center text-sm cursor-pointer text-muted-foreground', drag && 'border-primary bg-primary/5')}

        {upload.isPending ? 'Uploaden…' : 'Sleep een bestand hierheen of klik (max 20 MB)'}

      </div>
      <div className="flex justify-end"><Button size="sm" variant="ghost" onClick={onDone}>Annuleren</Button></div>
    </div>
  );
}

export function PrepIntakeTab({ companyId }: { companyId: string | null }) {
  const { data: items = [], isLoading } = useCompanyItems(companyId);
  const [mode, setMode] = useState<'idle' | 'choose' | 'note' | 'doc'>('idle');

  if (!companyId) return <p className="text-sm text-muted-foreground">Koppel eerst een bedrijf (links) om voorbereiding en intake vast te leggen.</p>;

          </section>
        );
      })}
    </div>
  );
}
```

```tsx
export const fmtWhen = (s: string) => new Date(s).toLocaleString('nl', { weekday: 'short', day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' });

export function AppointmentActions({ appt }: { appt: Appointment }) {
  const { user } = useAuth();
  const complete = useCompleteAppointment();
  const cancel = useCancelAppointment();
  const [outcome, setOutcome] = useState('');
  const [resched, setResched] = useState(false);
  const ics = () => downloadIcs({
    uid: appt.id, title: appt.title, start: new Date(appt.starts_at), durationMin: appt.duration_min,
    description: [appt.notes, appt.deal_id && `${window.location.origin}/pipeline?deal=${appt.deal_id}`].filter(Boolean).join('\n'),
    location: appt.location, organizerEmail: appt.owner_id === user?.id ? user?.email : null,
    attendeeEmail: appt.contact?.email ?? null, attendeeName: appt.contact?.name ?? null,
  });
  return (
    <div className="flex flex-wrap gap-1 mt-1.5">
      <Button size="sm" variant="ghost" className="h-7 px-2 text-xs gap-1" onClick={ics}><CalendarPlus className="h-3.5 w-3.5" /> Outlook</Button>
      <Popover>
        <PopoverTrigger asChild><Button size="sm" variant="ghost" className="h-7 px-2 text-xs gap-1"><Check className="h-3.5 w-3.5" /> Afronden</Button></PopoverTrigger>
        <PopoverContent className="w-72 space-y-2">
          <Textarea placeholder="Uitkomst (kort)" value={outcome} onChange={e => setOutcome(e.target.value)} className="min-h-[70px]" />
          <div className="flex justify-end gap-2">
            <Button size="sm" variant="outline" onClick={() => complete.mutate({ appt, outcome, status: 'no_show' })}>No-show</Button>
            <Button size="sm" onClick={() => complete.mutate({ appt, outcome, status: 'done' })}>Gedaan</Button>
          </div>
        </PopoverContent>
      </Popover>
      <Button size="sm" variant="ghost" className="h-7 px-2 text-xs gap-1" onClick={() => setResched(true)}><CalendarClock className="h-3.5 w-3.5" /> Verzetten</Button>
      <Button size="sm" variant="ghost" className="h-7 px-2 text-xs gap-1 text-muted-foreground hover:text-destructive" onClick={() => cancel.mutate(appt)}><X className="h-3.5 w-3.5" /> Annuleren</Button>
      <AppointmentDialog open={resched} onClose={() => setResched(false)} deal={null} type={appt.type} appointment={appt} />
    </div>
  );
}
```

```ts
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

// null = allowed. Allows an admin user's JWT, or the service-role key (cron/server calls).
export async function requireAdmin(req: Request, cors: Record<string, string>): Promise<Response | null> {
  const deny = (status: number, error: string) =>
    new Response(JSON.stringify({ error }), { status, headers: { ...cors, "Content-Type": "application/json" } });
  const auth = req.headers.get("Authorization") ?? "";
  const token = auth.replace(/^Bearer\s+/i, "");
  if (!token) return deny(401, "Unauthorized");
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
  if (token === serviceKey) return null;
  const url = Deno.env.get("SUPABASE_URL")!;
  const userClient = createClient(url, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: auth } } });
  const { data: { user } } = await userClient.auth.getUser();
  if (!user) return deny(401, "Unauthorized");
  const admin = createClient(url, serviceKey);
  const { data: isAdmin } = await admin.rpc("is_admin", { _user_id: user.id });
  return isAdmin ? null : deny(403, "Forbidden: admin only");
}
```
